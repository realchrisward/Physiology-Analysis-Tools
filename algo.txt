def beatepocher_kde_clipped_rr_smooth(
    filtered_data_frame, beat_df, voltage_column="ecg", time_column="time", beat_window=1, bias=0,
    n_decimals=TIMESTAMP_DECIMALS, beat_length=128, kde_bandwidth=0.05, min_rr=0.1, max_rr=0.1667
):
    """
    beatepocher_v8 but with kde_bandwidth default changed to 0.05 for smoother distribution.
    """
    ts = beat_df["ts"].values
    ts_diff_all = np.diff(ts)

    def extract_first_modality_kde_robust(rr_data, bandwidth=0.05):
        if len(rr_data) < 2:
            return rr_data
        # Ignore lowest RR intervals (potential noise)
        kde = gaussian_kde(rr_data, bw_method=bandwidth)
        rr_grid = np.linspace(np.min(rr_data), np.max(rr_data), 1000)
        kde_vals = kde(rr_grid)
        peaks, _ = find_peaks(kde_vals)
        if len(peaks) == 0:
            return rr_data
        
        # Choose the biggest peak as the main peak. This is more robust than just taking the first peak since there can be false detection leading to peaks in the left.
        main_peak = peaks[np.argmax(kde_vals[peaks])]

        # Detects all local minima in the KDE curve by finding points where the second derivative changes sign
        minima = (np.diff(np.sign(np.diff(kde_vals))) > 0).nonzero()[0] + 1  
        
        # Selects only the minima that occur to the right of the main peak (i.e., larger RR values).
        right_min = minima[minima > main_peak]
        if len(right_min) > 0:
            # Takes the first such minimum, which represents the boundary between the first mode and subsequent modes.
            cutoff_idx = right_min[0]
            # Converts this index into the actual RR interval value at the minimum.
            cutoff_val = rr_grid[cutoff_idx]
            # Keeps only RR values belonging to the first modality, i.e., those to the left of the cutoff.
            modality_mask = rr_data < cutoff_val
            return rr_data[modality_mask]
        else:
            return rr_data

    if len(ts_diff_all) == 0:
        rr_window = 0
    else:
        rr_first_modality = extract_first_modality_kde_robust(ts_diff_all, bandwidth=kde_bandwidth)
        median_rr = np.median(rr_first_modality) if len(rr_first_modality) > 0 else np.median(ts_diff_all)
        rr_window = np.clip(median_rr, min_rr, max_rr)
    sample_rate = 1 / (filtered_data_frame[time_column].iloc[1] - filtered_data_frame[time_column].iloc[0])
    r_bias = (1 + bias) / 2
    l_bias = 1 - r_bias
    left_window = l_bias * beat_window * rr_window * sample_rate
    right_window = r_bias * beat_window * rr_window * sample_rate
    epochs_dict = {}
    for idx, ts_val in enumerate(beat_df.ts):
        index = list(filtered_data_frame[time_column]).index(ts_val)
        start = int(index - round(left_window))
        end = int(index + round(right_window))
        if (start < min(filtered_data_frame.index)) or (end > max(filtered_data_frame.index)):
            continue
        data_epoch = filtered_data_frame.loc[start:end]
        beat_epoch = data_epoch[voltage_column].to_numpy()
        beat_epoch = sp.signal.resample(beat_epoch, beat_length)
        _store_rounded_timestamp(
            epochs_dict,
            ts_val,
            ml_tools.detrend_normalise(beat_epoch),
            n_decimals,
            context="beatepocher_kde_clipped_rr_smooth",
        )
    return epochs_dict
